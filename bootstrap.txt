val Obsidian = Color(0xFF05060A); val Navy = Color(0xFF0B1426); val Cyan = Color(0xFF00E5FF); val Violet = Color(0xFF9B5CFF)

class MainActivity : ComponentActivity() {
    override fun onCreate(s: Bundle?) {
        super.onCreate(s)
        requestPermissions(arrayOf(Manifest.permission.POST_NOTIFICATIONS), 1)
        setContent { App() }
    }
}

@Composable
fun App() {
    val ctx = LocalContext.current
    var tab by remember { mutableStateOf(0) }
    var on by remember { mutableStateOf(Prefs.on(ctx)) }
    var dim by remember { mutableStateOf(Prefs.dim(ctx)) }
    var accentIdx by remember { mutableStateOf(Prefs.accent(ctx)) }
    val accent = if (accentIdx == 0) Cyan else Violet

    fun apply(enable: Boolean) {
        if (enable && !Settings.canDrawOverlays(ctx)) {
            ctx.startActivity(Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION, Uri.parse("package:${ctx.packageName}")))
            return
        }
        val i = Intent(ctx, OverlayService::class.java)
        if (enable) ctx.startForegroundService(i) else ctx.stopService(i)
        on = enable
    }

    Column(Modifier.fillMaxSize().background(Brush.verticalGradient(listOf(Obsidian, Navy))).statusBarsPadding()) {
        Column(Modifier.weight(1f).padding(24.dp), horizontalAlignment = Alignment.CenterHorizontally) {
            Text("PrivacyGuard", color = Color.White, fontSize = 28.sp, fontWeight = FontWeight.Bold)
            Text("Your screen stays yours.", color = accent, fontSize = 14.sp)
            Spacer(Modifier.height(24.dp))
            when (tab) {
                0 -> {
                    Box(Modifier.size(160.dp).clip(CircleShape).background(Brush.radialGradient(listOf(accent.copy(alpha = if (on) .5f else .15f), Color.Transparent)))
                        .border(2.dp, accent, CircleShape), contentAlignment = Alignment.Center) { Image(painterResource(R.drawable.ic_launcher_foreground), contentDescription = "PrivacyGuard", modifier = Modifier.size(150.dp)) }
                    Spacer(Modifier.height(20.dp))
                    Text(if (on) "Protection ON" else "Protection OFF", color = Color.White, fontSize = 22.sp, fontWeight = FontWeight.SemiBold)
                    Spacer(Modifier.height(12.dp))
                    Switch(checked = on, onCheckedChange = { apply(it) },
                        colors = SwitchDefaults.colors(checkedTrackColor = accent, checkedThumbColor = Obsidian))
                    Spacer(Modifier.height(12.dp))
                    Text("Add the PrivacyGuard tile in Quick Settings for instant toggling.", color = Color.Gray, fontSize = 12.sp)
                }
                1 -> Card("Screen dimming") {
                    Text("Strength ${(dim * 100).toInt()}%", color = Color.White)
                    Slider(value = dim, onValueChange = { dim = it; Prefs.setDim(ctx, it) },
                        onValueChangeFinished = { if (on) apply(true) }, valueRange = 0.1f..0.85f,
                        colors = SliderDefaults.colors(thumbColor = accent, activeTrackColor = accent))
                    Text("Software can dim your whole screen, which makes it harder for anyone to read it. It can't darken only the side angles. That needs a privacy screen protector or a compatible display.", color = Color.Gray, fontSize = 12.sp)
                }
                2 -> Card("Accent color") {
                    Row(horizontalArrangement = Arrangement.spacedBy(16.dp)) {
                        listOf(Cyan, Violet).forEachIndexed { i, c ->
                            Box(Modifier.size(48.dp).clip(CircleShape).background(c)
                                .border(if (accentIdx == i) 3.dp else 0.dp, Color.White, CircleShape)
                                .clickable { accentIdx = i; Prefs.setAccent(ctx, i) })
                        }
                    }
                }
                else -> Card("Permissions") {
                    Text("Display over other apps: " + if (Settings.canDrawOverlays(ctx)) "granted" else "needed", color = Color.White)
                    Spacer(Modifier.height(8.dp))
                    Button(onClick = { ctx.startActivity(Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION, Uri.parse("package:${ctx.packageName}"))) },
                        colors = ButtonDefaults.buttonColors(containerColor = accent, contentColor = Obsidian)) { Text("Open permission settings") }
                    Spacer(Modifier.height(8.dp))
                    Text("No camera, no AI, no tracking, no cloud.", color = Color.Gray, fontSize = 12.sp)
                }
            }
        }
        NavigationBar(containerColor = Obsidian) {
            listOf("Home", "Protection", "Appearance", "Settings").forEachIndexed { i, t ->
                NavigationBarItem(selected = tab == i, onClick = { tab = i }, icon = { Text(listOf("🏠", "🛡", "🎨", "⚙")[i]) },
                    label = { Text(t) }, colors = NavigationBarItemDefaults.colors(selectedTextColor = accent, indicatorColor = Navy))
            }
        }
    }
}

@Composable
fun Card(title: String, content: @Composable ColumnScope.() -> Unit) {
    Column(Modifier.fillMaxWidth().clip(RoundedCornerShape(24.dp)).background(Navy).padding(20.dp)) {
        Text(title, color = Color.White, fontSize = 18.sp, fontWeight = FontWeight.SemiBold)
        Spacer(Modifier.height(12.dp)); content()
    }
}

package com.privacyguard.app
import android.app.*
import android.content.Context
import android.content.Intent
import android.graphics.Color
import android.graphics.PixelFormat
import android.os.IBinder
import android.view.View
import android.view.WindowManager

class OverlayService : Service() {
    private var view: View? = null
    override fun onBind(i: Intent?): IBinder? = null

    override fun onStartCommand(i: Intent?, f: Int, id: Int): Int {
        val nm = getSystemService(NotificationManager::class.java)
        nm.createNotificationChannel(NotificationChannel("pg", "PrivacyGuard", NotificationManager.IMPORTANCE_LOW))
        val pi = PendingIntent.getActivity(this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
        val n = Notification.Builder(this, "pg").setContentTitle("PrivacyGuard is on")
            .setContentText("Tap to adjust").setSmallIcon(android.R.drawable.ic_lock_lock).setContentIntent(pi).build()
        startForeground(1, n)
        if (!android.provider.Settings.canDrawOverlays(this)) { stopSelf(); return START_NOT_STICKY }
        val wm = getSystemService(Context.WINDOW_SERVICE) as WindowManager
        val v = view ?: View(this).also {
            it.setBackgroundColor(Color.BLACK)
            val lp = WindowManager.LayoutParams(-1, -1, WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
                PixelFormat.TRANSLUCENT)
            wm.addView(it, lp); view = it
        }
        v.alpha = Prefs.dim(this)
        Prefs.setOn(this, true)
        return START_STICKY
    }

    override fun onDestroy() {
        view?.let { (getSystemService(Context.WINDOW_SERVICE) as WindowManager).removeView(it) }
        view = null
        Prefs.setOn(this, false)
        super.onDestroy()
    }
}

package com.privacyguard.app
import android.content.Context
object Prefs {
    private fun p(c: Context) = c.getSharedPreferences("pg", Context.MODE_PRIVATE)
    fun on(c: Context) = p(c).getBoolean("on", false)
    fun setOn(c: Context, v: Boolean) = p(c).edit().putBoolean("on", v).apply()
    fun dim(c: Context) = p(c).getFloat("dim", 0.55f)
    fun setDim(c: Context, v: Float) = p(c).edit().putFloat("dim", v).apply()
    fun accent(c: Context) = p(c).getInt("accent", 0)
    fun setAccent(c: Context, v: Int) = p(c).edit().putInt("accent", v).apply()
}

package com.privacyguard.app
import android.content.Intent
import android.service.quicksettings.Tile
import android.service.quicksettings.TileService

class PrivacyTile : TileService() {
    override fun onStartListening() = refresh()
    override fun onClick() {
        val i = Intent(this, OverlayService::class.java)
        if (Prefs.on(this)) stopService(i) else startForegroundService(i)
        Prefs.setOn(this, !Prefs.on(this)); refresh()
    }
    private fun refresh() {
        qsTile?.let { it.state = if (Prefs.on(this)) Tile.STATE_ACTIVE else Tile.STATE_INACTIVE; it.updateTile() }
    }
}
